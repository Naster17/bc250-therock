cmake_minimum_required(VERSION 3.28)
project(bc250_ort_migraphx_admission LANGUAGES C CXX)
find_package(hip REQUIRED)
find_package(migraphx REQUIRED)
include(CheckSymbolExists)
set(CMAKE_REQUIRED_LIBRARIES migraphx::c)
check_symbol_exists(migraphx_onnx_options_set_external_data_path "migraphx/migraphx.h" HAVE_MIGRAPHX_API_ONNX_OPTIONS_SET_EXTERNAL_DATA_PATH)
if(NOT HAVE_MIGRAPHX_API_ONNX_OPTIONS_SET_EXTERNAL_DATA_PATH)
  message(FATAL_ERROR "ORT optional external-data API was not found")
endif()
add_executable(admission_probe probe.cpp api_symbols.cpp)
set_target_properties(admission_probe PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
target_compile_definitions(admission_probe PRIVATE __HIP_PLATFORM_AMD__=1)
target_link_libraries(admission_probe PRIVATE migraphx::c hip::host)

target_include_directories(admission_probe BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
foreach(standard 17 20)
  add_executable(header_cpp${standard} header_test.cpp)
  set_target_properties(header_cpp${standard} PROPERTIES CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES)
  target_include_directories(header_cpp${standard} BEFORE PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(header_cpp${standard} PRIVATE migraphx::c)
endforeach()
